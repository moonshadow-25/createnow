```python
from abc import ABC, abstractmethod
import httpx
import logging

logger = logging.getLogger(__name__)

class TTSService(ABC):
    """TTS 服务基类"""

    def __init__(self, api_url: str, api_key: str, model: str, project_id: str):
        self.api_url = api_url
        self.api_key = api_key
        self.model = model
        self.project_id = project_id
        self.client = httpx.AsyncClient(timeout=60.0)

    @abstractmethod
    async def generate(self, text: str, voice_id: str = None) -> dict:
        """
        生成音频

        返回格式:
        {
            "success": True/False,
            "audio_url": "https://...",  # 远程 URL（如果立即返回）
            "task_id": "...",  # 异步任务 ID（如果需要轮询）
            "error": "..."  # 错误信息
        }
        """
        pass

    @abstractmethod
    async def poll_task(self, task_id: str) -> dict:
        """
        轮询异步任务状态

        返回格式:
        {
            "status": "pending" | "completed" | "failed",
            "audio_url": "https://...",  # 完成后的 URL
            "error": "..."
        }
        """
        pass

    async def close(self):
        """关闭 HTTP 客户端"""
        await self.client.aclose()
```

```python
from app.services.ai.tts import TTSService
import logging

logger = logging.getLogger(__name__)

class OpenAITTSService(TTSService):
    """OpenAI TTS 适配器"""

    async def generate(self, text: str, voice_id: str = None) -> dict:
        """调用 OpenAI TTS API"""
        try:
            voice = voice_id or "alloy"  # 默认音色

            response = await self.client.post(
                f"{self.api_url}/audio/speech",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json={
                    "model": self.model,
                    "input": text,
                    "voice": voice,
                    "response_format": "mp3"
                }
            )

            if response.status_code == 200:
                # OpenAI 直接返回音频文件流
                # 需要先保存到临时文件，再上传到存储
                return {
                    "success": True,
                    "audio_data": response.content,  # 二进制数据
                    "format": "mp3"
                }
            else:
                return {
                    "success": False,
                    "error": f"API error: {response.text}"
                }
        except Exception as e:
            logger.error(f"OpenAI TTS error: {e}")
            return {"success": False, "error": str(e)}

    async def poll_task(self, task_id: str) -> dict:
        """OpenAI TTS 无需轮询"""
        return {"status": "completed"}
```

```python
from app.services.ai.tts import TTSService
import logging

logger = logging.getLogger(__name__)

class DashScopeTTSService(TTSService):
    """阿里云 DashScope TTS 适配器"""

    async def generate(self, text: str, voice_id: str = None) -> dict:
        """调用阿里云 TTS API"""
        try:
            voice = voice_id or "zhixiaobai"  # 默认音色

            response = await self.client.post(
                self.api_url,
                headers={
                    "Authorization": f"Bearer {self.api_key}",
                    "Content-Type": "application/json"
                },
                json={
                    "model": self.model,
                    "input": {"text": text},
                    "parameters": {
                        "voice": voice,
                        "format": "mp3"
                    }
                }
            )

            if response.status_code == 200:
                data = response.json()

                # 阿里云可能是同步返回或异步任务
                if "output" in data and "audio_url" in data["output"]:
                    # 同步返回
                    return {
                        "success": True,
                        "audio_url": data["output"]["audio_url"]
                    }
                elif "task_id" in data:
                    # 异步任务
                    return {
                        "success": True,
                        "task_id": data["task_id"]
                    }

            return {
                "success": False,
                "error": f"API error: {response.text}"
            }
        except Exception as e:
            logger.error(f"DashScope TTS error: {e}")
            return {"success": False, "error": str(e)}

    async def poll_task(self, task_id: str) -> dict:
        """轮询阿里云 TTS 任务"""
        try:
            response = await self.client.get(
                f"{self.api_url}/{task_id}",
                headers={"Authorization": f"Bearer {self.api_key}"}
            )

            if response.status_code == 200:
                data = response.json()
                status = data.get("status", "pending")

                if status == "succeeded":
                    return {
                        "status": "completed",
                        "audio_url": data["output"]["audio_url"]
                    }
                elif status == "failed":
                    return {
                        "status": "failed",
                        "error": data.get("message", "Unknown error")
                    }
                else:
                    return {"status": "pending"}

            return {"status": "failed", "error": response.text}
        except Exception as e:
            logger.error(f"DashScope poll error: {e}")
            return {"status": "failed", "error": str(e)}
```

```python
import json
import logging
from pathlib import Path
from typing import List, Optional
from app.core.config import settings

logger = logging.getLogger(__name__)

class AudioService:
    """音频服务"""

    @staticmethod
    def save_audio_record(project_id: str, audio_id: str, record: dict) -> None:
        """保存音频记录"""
        audios_dir = settings.PROJECTS_DIR / project_id / "audios"
        audios_dir.mkdir(exist_ok=True)

        audio_file = audios_dir / f"{audio_id}.json"
        with open(audio_file, "w", encoding="utf-8") as f:
            json.dump(record, f, ensure_ascii=False, indent=2)

    @staticmethod
    def get_audio(project_id: str, audio_id: str) -> Optional[dict]:
        """获取单个音频记录"""
        audio_file = settings.PROJECTS_DIR / project_id / "audios" / f"{audio_id}.json"
        if not audio_file.exists():
            return None

        with open(audio_file, "r", encoding="utf-8") as f:
            return json.load(f)

    @staticmethod
    def list_audios(project_id: str, asset_id: str = None, asset_type: str = None) -> List[dict]:
        """列出音频记录"""
        audios_dir = settings.PROJECTS_DIR / project_id / "audios"
        if not audios_dir.exists():
            return []

        audios = []
        for audio_file in audios_dir.glob("*.json"):
            try:
                with open(audio_file, "r", encoding="utf-8") as f:
                    audio = json.load(f)

                    # 过滤条件
                    if asset_id and audio.get("asset_id") != asset_id:
                        continue
                    if asset_type and audio.get("asset_type") != asset_type:
                        continue

                    audios.append(audio)
            except Exception as e:
                logger.error(f"Error reading audio file {audio_file}: {e}")

        # 按创建时间倒序
        audios.sort(key=lambda x: x.get("created_at", ""), reverse=True)
        return audios

    @staticmethod
    def set_primary_audio(project_id: str, audio_id: str, asset_id: str) -> dict:
        """设置主音频"""
        audios_dir = settings.PROJECTS_DIR / project_id / "audios"

        # 取消同一 asset 的其他音频的主音频状态
        for audio_file in audios_dir.glob("*.json"):
            try:
                with open(audio_file, "r", encoding="utf-8") as f:
                    audio = json.load(f)

                if audio.get("asset_id") == asset_id and audio.get("is_primary"):
                    audio["is_primary"] = False
                    with open(audio_file, "w", encoding="utf-8") as f:
                        json.dump(audio, f, ensure_ascii=False, indent=2)
            except Exception as e:
                logger.error(f"Error updating audio {audio_file}: {e}")

        # 设置目标音频为主音频
        target_file = audios_dir / f"{audio_id}.json"
        with open(target_file, "r", encoding="utf-8") as f:
            audio = json.load(f)

        audio["is_primary"] = True
        with open(target_file, "w", encoding="utf-8") as f:
            json.dump(audio, f, ensure_ascii=False, indent=2)

        return audio

    @staticmethod
    def delete_audio(project_id: str, audio_id: str) -> None:
        """删除音频"""
        audio_file = settings.PROJECTS_DIR / project_id / "audios" / f"{audio_id}.json"
        if audio_file.exists():
            # 删除本地文件
            audio = AudioService.get_audio(project_id, audio_id)
            if audio and audio.get("local_path"):
                local_file = settings.PROJECTS_DIR / project_id / "audios" / "files" / audio["local_path"]
                if local_file.exists():
                    local_file.unlink()

            # 删除记录
            audio_file.unlink()
```

```python
import httpx
import logging
from pathlib import Path
from app.core.config import settings

logger = logging.getLogger(__name__)

class AudioDownloadService:
    """音频下载服务"""

    @staticmethod
    async def download_and_save_audio(
        project_id: str,
        audio_id: str,
        url: str,
        asset_type: str = "storyboard"
    ) -> str:
        """
        下载音频并保存到本地

        返回: local_path (相对路径，如 "storyboard/audio_xxx.mp3")
        """
        try:
            # 下载音频
            async with httpx.AsyncClient(timeout=120.0) as client:
                response = await client.get(url)
                response.raise_for_status()
                audio_data = response.content

            # 确定文件名
            filename = f"{audio_id}.mp3"
            local_path = f"{asset_type}/{filename}"

            # 保存到本地
            save_dir = settings.PROJECTS_DIR / project_id / "audios" / "files" / asset_type
            save_dir.mkdir(parents=True, exist_ok=True)

            save_file = save_dir / filename
            with open(save_file, "wb") as f:
                f.write(audio_data)

            logger.info(f"[音频下载] 下载成功: {local_path}")

            # 更新音频记录
            from app.services.audio_service import AudioService
            audio = AudioService.get_audio(project_id, audio_id)
            if audio:
                audio["local_path"] = local_path
                AudioService.save_audio_record(project_id, audio_id, audio)

            return local_path

        except Exception as e:
            logger.error(f"[音频下载] 下载失败 {audio_id}: {e}")
            raise
```

```python
import logging
import uuid
from datetime import datetime
from fastapi import APIRouter, HTTPException, Body

from app.services import get_ai_service
from app.services.audio_service import AudioService
from app.core.config import settings

logger = logging.getLogger(__name__)

router = APIRouter()

@router.post("/audio")
async def generate_audio(project_id: str, request: dict):
    """生成音频"""
    from app.services import ProjectService

    project = ProjectService.get_project(project_id)
    if not project:
        raise HTTPException(status_code=404, detail="Project not found")

    ai_config = project.get("ai_config", {})
    tts_service = get_ai_service(ai_config, "tts", project_id)

    text = request.get("text", "")
    voice_id = request.get("voice_id")
    asset_id = request.get("asset_id", "")
    asset_type = request.get("asset_type", "storyboard_dialogue")

    if not text:
        raise HTTPException(status_code=400, detail="Text is required")

    try:
        result = await tts_service.generate(text=text, voice_id=voice_id)
        await tts_service.close()

        if not result.get("success"):
            raise HTTPException(status_code=500, detail=result.get("error", "TTS generation failed"))

        # 创建音频记录
        audio_id = str(uuid.uuid4())
        record = {
            "audio_id": audio_id,
            "asset_id": asset_id,
            "asset_type": asset_type,
            "text": text,
            "voice_id": voice_id,
            "audio_path": result.get("audio_url"),
            "task_id": result.get("task_id"),
            "status": "completed" if result.get("audio_url") else "pending",
            "model": ai_config.get("tts", {}).get("model", ""),
            "created_at": datetime.now().isoformat(),
        }

        # 如果是直接返回的音频数据（OpenAI），先保存
        if result.get("audio_data"):
            from app.services.audio_download_service import AudioDownloadService
            # 保存二进制数据
            save_dir = settings.PROJECTS_DIR / project_id / "audios" / "files" / asset_type
            save_dir.mkdir(parents=True, exist_ok=True)
            filename = f"{audio_id}.mp3"
            save_file = save_dir / filename
            with open(save_file, "wb") as f:
                f.write(result["audio_data"])
            record["local_path"] = f"{asset_type}/{filename}"
            record["status"] = "completed"

        AudioService.save_audio_record(project_id, audio_id, record)

        # 如果有URL，异步下载
        if result.get("audio_url"):
            import asyncio
            from app.services.audio_download_service import AudioDownloadService
            asyncio.create_task(
                AudioDownloadService.download_and_save_audio(
                    project_id=project_id,
                    audio_id=audio_id,
                    url=result["audio_url"],
                    asset_type=asset_type
                )
            )

        return record

    except HTTPException:
        raise
    except Exception as e:
        await tts_service.close()
        logger.error(f"Audio generation error: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/audios/{audio_id}/poll")
async def poll_audio_status(project_id: str, audio_id: str):
    """轮询音频状态"""
    from app.services import ProjectService

    audio = AudioService.get_audio(project_id, audio_id)
    if not audio:
        raise HTTPException(status_code=404, detail="Audio not found")

    if audio.get("status") in ("completed", "failed"):
        return audio

    task_id = audio.get("task_id")
    if not task_id:
        raise HTTPException(status_code=400, detail="No task_id found")

    project = ProjectService.get_project(project_id)
    ai_config = project.get("ai_config", {})
    tts_service = get_ai_service(ai_config, "tts", project_id)

    try:
        result = await tts_service.poll_task(task_id)
        await tts_service.close()

        # 更新记录
        audio["poll_count"] = audio.get("poll_count", 0) + 1
        audio["last_poll_time"] = datetime.now().isoformat()

        if result.get("status") == "completed":
            audio["status"] = "completed"
            audio["audio_path"] = result.get("audio_url")

            # 异步下载
            if result.get("audio_url"):
                import asyncio
                from app.services.audio_download_service import AudioDownloadService
                asyncio.create_task(
                    AudioDownloadService.download_and_save_audio(
                        project_id=project_id,
                        audio_id=audio_id,
                        url=result["audio_url"],
                        asset_type=audio.get("asset_type", "storyboard")
                    )
                )
        elif result.get("status") == "failed":
            audio["status"] = "failed"
            audio["error"] = result.get("error")

        AudioService.save_audio_record(project_id, audio_id, audio)
        return audio

    except Exception as e:
        await tts_service.close()
        logger.error(f"Poll audio error: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

## 🎯 上下文 3：主体引用系统和虚拟主体 ⚠️ 【重点详解】

### 一、核心概念

#### 1.1 什么是主体引用？
主体引用是指在视频生成时，让 AI 视频模型知道画面中应该出现哪些特定的角色（主体）。

**两种实现方式：**
- **真实主体**：模型直接支持主体引用（如 Sora），通过 `subject_id` 告诉模型使用哪个角色
- **虚拟主体**：模型不支持主体引用，通过替换提示词中的占位符来描述角色

#### 1.2 虚拟主体的工作流程

```
用户操作流程:
1. 用户勾选"引用主体" ✓
2. 系统自动将角色添加到提示词中（以标签形式）
3. 用户编辑提示词（可删除/移动标签）
4. 用户点击"生成视频"

后端处理流程（虚拟主体模式）:
1. 检测视频模型是否支持主体引用
2. 如果不支持，启动虚拟主体流程：
   a. 调用 VLM 分析每个角色的图片
   b. VLM 返回描述文本（如"年轻的男子"）
   c. 将提示词中的 "{孙权}" 替换为 "年轻的男子"
   d. 将处理后的提示词发送给视频 API
3. 如果支持，直接传递 subject_ids
```

#### 1.3 占位符格式

**PRD 原文分析：**
> "虚拟主体通过提示词描述，通过自动替换提示词的中关于主体的描述来实现"
> "生成类似——'主体1'做了什么事情，'主体2'做了什么事情"
> "自动将'主体1'、'主体2'替换成符合图片中的描述，而不是主体的名字"

**我的理解（需要你确认）：**

有两种占位符格式：
- **格式 A - 角色名**：`{孙权}`、`{刘备}` - 用户友好，直观
- **格式 B - 序号**：`{主体1}`、`{主体2}` - 抽象，通用

**我的疑问 ❓：**
1. 两种格式是否都支持？还是只用其中一种？
2. 如果两种都支持，什么时候用哪种？
3. 占位符是用大括号 `{}` 还是其他符号？

**我的建议（待确认）：**
- 统一使用角色名格式：`{孙权}`、`{刘备}`
- 理由：用户友好，提示词可读性强
- 在后端内部映射为 `主体1`、`主体2` 等

```python
@staticmethod
async def build_subject_refs(
    project_id: str,
    storyboard: dict,
    vlm_service = None
) -> List[dict]:
    """
    为分镜构建主体引用列表

    步骤:
    1. 从 storyboard.character_ids 获取所有角色
    2. 加载每个角色的详细信息
    3. 判断角色是否有 subject_id（真实主体）
    4. 如果没有 subject_id，使用 VLM 分析角色图片生成描述
    5. 构建 SubjectReference 对象
    """
    from app.services.asset_service import AssetService, ImageService

    character_ids = storyboard.get("character_ids", [])
    subject_refs = []

    for idx, character_id in enumerate(character_ids):
        # 加载角色
        character = AssetService.load_asset(project_id, "character", character_id)
        if not character:
            continue

        # 获取角色图片
        image_url = None
        if character.get("image_id"):
            image = ImageService.get_image(project_id, character["image_id"])
            if image:
                # 优先使用本地图片
                if image.get("local_path"):
                    local_file = settings.PROJECTS_DIR / project_id / "images" / "files" / image["local_path"]
                    if local_file.exists():
                        from app.services.image_download_service import ImageDownloadService
                        image_url = ImageDownloadService.image_to_base64_url(local_file)

                # 降级到远程 URL
                if not image_url and image.get("image_path"):
                    image_url = image["image_path"]

        # 构建占位符
        character_name = character.get("name", f"角色{idx+1}")
        placeholder = f"{{{character_name}}}"  # 格式: {孙权}

        # 判断是否有真实主体
        subject_id = character.get("subject_id")
        description = None

        # 如果没有真实主体，使用 VLM 生成描述
        if not subject_id and vlm_service and image_url:
            description = await SubjectService.analyze_subject_with_vlm(
                vlm_service=vlm_service,
                image_url=image_url,
                character_name=character_name
            )

        subject_refs.append({
            "character_id": character_id,
            "character_name": character_name,
            "subject_id": subject_id,
            "placeholder": placeholder,
            "description": description,
            "image_url": image_url,
            "order": idx + 1
        })

    return subject_refs
```

```python
def replace_subject_placeholders(
    prompt: str,
    subject_refs: List[dict],
    use_real_subjects: bool
) -> str:
    """
    替换提示词中的主体占位符

    参数:
    - prompt: 原始提示词（如 "{孙权}和{刘备}在宫殿中讨论军情"）
    - subject_refs: 主体引用列表
    - use_real_subjects: 是否使用真实主体（模型支持主体引用）

    返回:
    - 处理后的提示词
      - 真实主体模式: "{孙权}和{刘备}在宫殿中讨论军情" (不变)
      - 虚拟主体模式: "年轻的男子和中年男子在宫殿中讨论军情"
    """
    if use_real_subjects:
        # 真实主体模式：保留占位符
        return prompt

    # 虚拟主体模式：替换占位符为描述
    result = prompt
    for ref in subject_refs:
        placeholder = ref["placeholder"]  # "{孙权}"
        description = ref.get("description", ref["character_name"])  # "年轻的男子"

        # 替换所有出现的占位符
        result = result.replace(placeholder, description)

    logger.info(f"[虚拟主体] 原提示词: {prompt}")
    logger.info(f"[虚拟主体] 新提示词: {result}")

    return result
```

```typescript
import React, { useState, useEffect } from 'react';
import { Tag, Space, Button } from 'antd';
import { SubjectReference } from '../../types';

interface SubjectTagInputProps {
  prompt: string;
  subjectRefs: SubjectReference[];
  onPromptChange: (prompt: string) => void;
  showDescriptions?: boolean;  // 是否显示虚拟主体描述
}

export const SubjectTagInput: React.FC<SubjectTagInputProps> = ({
  prompt,
  subjectRefs,
  onPromptChange,
  showDescriptions = false,
}) => {
  // 解析提示词，提取标签和文本
  const parsePrompt = (text: string) => {
    const parts: Array<{ type: 'tag' | 'text'; content: string }> = [];
    const regex = /\{([^}]+)\}/g;  // 匹配 {xxx}
    let lastIndex = 0;
    let match;

    while ((match = regex.exec(text)) !== null) {
      // 添加标签前的文本
      if (match.index > lastIndex) {
        parts.push({
          type: 'text',
          content: text.substring(lastIndex, match.index)
        });
      }

      // 添加标签
      parts.push({
        type: 'tag',
        content: match[1]  // 标签内容（如 "孙权"）
      });

      lastIndex = match.index + match[0].length;
    }

    // 添加最后的文本
    if (lastIndex < text.length) {
      parts.push({
        type: 'text',
        content: text.substring(lastIndex)
      });
    }

    return parts;
  };

  const parts = parsePrompt(prompt);

  // 删除标签
  const handleRemoveTag = (tagName: string) => {
    const newPrompt = prompt.replace(`{${tagName}}`, '');
    onPromptChange(newPrompt.trim());
  };

  // 从主体栏添加标签
  const handleAddTag = (ref: SubjectReference) => {
    // 检查是否已存在
    if (prompt.includes(ref.placeholder)) {
      return;
    }

    // 添加到提示词开头
    const newPrompt = `${ref.placeholder} ${prompt}`;
    onPromptChange(newPrompt);
  };

  // 获取未使用的主体
  const unusedSubjects = subjectRefs.filter(
    ref => !prompt.includes(ref.placeholder)
  );

  return (
    <div>
      {/* 主体栏 */}
      {unusedSubjects.length > 0 && (
        <div style={{ marginBottom: 8 }}>
          <span style={{ marginRight: 8 }}>主体栏:</span>
          <Space size="small">
            {unusedSubjects.map(ref => (
              <Button
                key={ref.character_id}
                size="small"
                type="dashed"
                onClick={() => handleAddTag(ref)}
              >
                + {ref.character_name}
              </Button>
            ))}
          </Space>
        </div>
      )}

      {/* 提示词编辑区 */}
      <div style={{
        border: '1px solid #d9d9d9',
        borderRadius: 4,
        padding: 8,
        minHeight: 100
      }}>
        {parts.map((part, idx) => {
          if (part.type === 'tag') {
            return (
              <Tag
                key={idx}
                closable
                onClose={() => handleRemoveTag(part.content)}
                color="blue"
                style={{ margin: 2 }}
              >
                {part.content}
              </Tag>
            );
          } else {
            return (
              <span key={idx} contentEditable suppressContentEditableWarning>
                {part.content}
              </span>
            );
          }
        })}
      </div>

      {/* 虚拟主体描述 */}
      {showDescriptions && subjectRefs.some(ref => ref.description) && (
        <div style={{ marginTop: 8, fontSize: 12, color: '#666' }}>
          <div>虚拟主体描述:</div>
          {subjectRefs
            .filter(ref => ref.description)
            .map(ref => (
              <div key={ref.character_id}>
                • {ref.character_name}: {ref.description}
              </div>
            ))}
        </div>
      )}
    </div>
  );
};
```

```python
    """导出项目到剪影（包含音频轨道）"""
    # ... 现有视频导出逻辑 ...

    # 【新增】音频轨道收集
    audio_tracks = []
    current_time = 0

    for storyboard in storyboards:
        storyboard_id = storyboard["asset_id"]

        # 获取主音频
        from app.services.audio_service import AudioService
        audios = AudioService.list_audios(
            project_id,
            asset_id=storyboard_id,
            asset_type="storyboard_dialogue"
        )

        primary_audio = next((a for a in audios if a.get("is_primary")), None)

        if primary_audio and primary_audio.get("local_path"):
            audio_file = settings.PROJECTS_DIR / project_id / "audios" / "files" / primary_audio["local_path"]

            if audio_file.exists():
                # 复制音频到导出目录
                audio_export_dir = Path(export_dir) / "audios"
                audio_export_dir.mkdir(exist_ok=True)

                audio_export_path = audio_export_dir / audio_file.name
                shutil.copy(audio_file, audio_export_path)

                # 记录音频轨道
                audio_tracks.append({
                    "sequence": storyboard["sequence"],
                    "audio_path": str(audio_export_path),
                    "duration": primary_audio.get("duration", 5.0),
                    "start_time": current_time,  # 以视频起始位置作为音频起始位置
                    "text": primary_audio.get("text", "")
                })

        # 累加时间（假设每个视频 5 秒）
        current_time += 5.0

    # 添加到导出数据
    project_data["audio_tracks"] = audio_tracks

    # ... 保存 jiaying_project.json ...
```

#### 1. 音频库组件
**新建文件**: `frontend/src/components/storyboard/AudioLibrary.tsx`

```typescript
import React, { useState, useEffect } from 'react';
import { Card, List, Button, Popconfirm, message } from 'antd';
import { PlayCircleOutlined, DeleteOutlined, StarOutlined, StarFilled } from '@ant-design/icons';
import { AudioGeneration } from '../../types';
import { storyboardApi, audioApi } from '../../services/api';

interface AudioLibraryProps {
  projectId: string;
  storyboardId: string;
  dialogue: string;
}

export const AudioLibrary: React.FC<AudioLibraryProps> = ({
  projectId,
  storyboardId,
  dialogue,
}) => {
  const [audios, setAudios] = useState<AudioGeneration[]>([]);
  const [loading, setLoading] = useState(false);
  const [generating, setGenerating] = useState(false);

  useEffect(() => {
    loadAudios();
  }, [projectId, storyboardId]);

  const loadAudios = async () => {
    setLoading(true);
    try {
      const { data } = await storyboardApi.getAudios(projectId, storyboardId);
      setAudios(data);
    } catch (error: any) {
      message.error(`加载音频库失败: ${error.message}`);
    } finally {
      setLoading(false);
    }
  };

  const handleGenerate = async () => {
    setGenerating(true);
    try {
      const { data } = await storyboardApi.generateAudio(
        projectId,
        storyboardId,
        dialogue,
        false
      );

      if (data.is_os) {
        message.success('音频生成任务已提交');
        loadAudios();
      } else {
        message.info(data.message);
      }
    } catch (error: any) {
      message.error(`生成失败: ${error.message}`);
    } finally {
      setGenerating(false);
    }
  };

  const handleSetPrimary = async (audioId: string) => {
    try {
      await audioApi.setPrimary(projectId, audioId, storyboardId);
      message.success('已设置为主音频');
      loadAudios();
    } catch (error: any) {
      message.error(`设置失败: ${error.message}`);
    }
  };

  const handleDelete = async (audioId: string) => {
    try {
      await audioApi.delete(projectId, audioId);
      message.success('已删除');
      loadAudios();
    } catch (error: any) {
      message.error(`删除失败: ${error.message}`);
    }
  };

  const handlePlay = (audio: AudioGeneration) => {
    const audioElement = new Audio();

    if (audio.local_path) {
      audioElement.src = `/api/projects/${projectId}/audios/files/${audio.local_path}`;
    } else if (audio.audio_path) {
      audioElement.src = audio.audio_path;
    } else {
      message.warning('音频文件不可用');
      return;
    }

    audioElement.play();
  };

  return (
    <Card
      title="🎵 音频库"
      size="small"
      extra={
        <Button
          type="primary"
          size="small"
          onClick={handleGenerate}
          loading={generating}
        >
          生成音频
        </Button>
      }
    >
      <List
        loading={loading}
        dataSource={audios}
        renderItem={(audio) => (
          <List.Item
            actions={[
              <Button
                type="text"
                icon={<PlayCircleOutlined />}
                onClick={() => handlePlay(audio)}
              >
                播放
              </Button>,
              <Button
                type="text"
                icon={audio.is_primary ? <StarFilled style={{ color: '#faad14' }} /> : <StarOutlined />}
                onClick={() => handleSetPrimary(audio.audio_id)}
              >
                {audio.is_primary ? '主音频' : '设为主音频'}
              </Button>,
              <Popconfirm
                title="确定删除此音频吗？"
                onConfirm={() => handleDelete(audio.audio_id)}
              >
                <Button type="text" danger icon={<DeleteOutlined />}>
                  删除
                </Button>
              </Popconfirm>,
            ]}
          >
            <List.Item.Meta
              title={`"${audio.text.substring(0, 30)}..."`}
              description={`时长: ${audio.duration || '未知'}s | 状态: ${audio.status}`}
            />
          </List.Item>
        )}
      />
    </Card>
  );
};
```
